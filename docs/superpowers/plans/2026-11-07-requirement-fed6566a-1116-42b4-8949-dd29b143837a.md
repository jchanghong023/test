**Goal:** Add a single Python standard-library script `machine_info.py` that prints CPU, memory, disk, OS, hostname, and architecture information to the terminal.

**Architecture:** One standalone script with platform-branched collectors (Windows via ctypes/registry/wmic, Linux via /proc, macOS best-effort). Human-readable sectioned output. No CLI args, no file writes, no third-party deps.

**Tech Stack:** Python 3 stdlib only (`platform`, `os`, `shutil`, `ctypes`, `subprocess`).

```python
import subprocess, sys, unittest

class TestMachineInfo(unittest.TestCase):
    def test_script_runs_and_contains_fields(self):
        r = subprocess.run([sys.executable, "machine_info.py"], capture_output=True, text=True)
        self.assertEqual(r.returncode, 0, r.stderr)
        for key in ["CPU", "Memory", "Disk", "OS", "Hostname", "Architecture"]:
            self.assertIn(key, r.stdout)

    def test_unknown_arg_rejected(self):
        r = subprocess.run([sys.executable, "machine_info.py", "--json"], capture_output=True, text=True)
        self.assertNotEqual(r.returncode, 0)
        self.assertNotIn("Traceback", r.stdout + r.stderr)

    def test_output_has_units(self):
        r = subprocess.run([sys.executable, "machine_info.py"], capture_output=True, text=True)
        self.assertEqual(r.returncode, 0)
        self.assertTrue(any(u in r.stdout for u in ("GB", "MB", "bytes")))

if __name__ == "__main__":
    unittest.main()
```
